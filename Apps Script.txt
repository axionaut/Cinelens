const MODEL = 'gemini-3.1-flash-lite';
const MAX_ITEMS = 20;
const MAX_STORY_CHARS = 14000;
const DEFAULT_MIN_TAGS = 10;
const DEFAULT_MAX_TAGS = 20;
const MAX_VOCABULARY_TAGS = 6000;

function doGet() {
  return jsonOutput({
    ok: true,
    service: 'CineLens Gemini Tagger',
    model: MODEL,
    supports: ['tag-titles', 'normalize-tag-cloud']
  });
}

function doPost(e) {
  try {
    const request = JSON.parse(
      e && e.postData && e.postData.contents || '{}'
    );

    if (
      request.task === 'normalize-tag-cloud' ||
      request.normalizeVocabularyOnly === true
    ) {
      return jsonOutput(handleTagCloudNormalization(request));
    }

    if (request.task === 'mood-titles') {
      return jsonOutput(handleMoodTagging(request));
    }

    return jsonOutput(handleTitleTagging(request));
  } catch (error) {
    return jsonOutput({
      ok: false,
      error: String(error && error.message || error)
    });
  }
}

function handleMoodTagging(request) {
  const items = Array.isArray(request.items) ? request.items.slice(0, MAX_ITEMS) : [];
  if (!items.length) throw new Error('No titles supplied');
  const cleanItems = items.map(function (item, index) {
    return {
      id: String(item.id || index),
      title: String(item.title || '').slice(0, 200),
      plot: String(item.storyText || '').slice(0, MAX_STORY_CHARS)
    };
  });
  const schema = {
    type: 'object',
    properties: {
      results: {
        type: 'array',
        items: {
          type: 'object',
          properties: {
            id: {type: 'string'},
            moods: {
              type: 'array',
              minItems: 1,
              maxItems: 1,
              items: {
                type: 'object',
                properties: {
                  mood: {type: 'string', enum: ['happy','sad','romantic','funny','scary','tense','calm','dark']},
                  confidence: {type: 'number', minimum: 0, maximum: 1},
                  evidence: {type: 'string'}
                },
                required: ['mood', 'confidence', 'evidence'],
                additionalProperties: false
              }
            }
          },
          required: ['id', 'moods'],
          additionalProperties: false
        }
      }
    },
    required: ['results'],
    additionalProperties: false
  };
  const prompt = [
    'You classify the emotional viewing mood of movie and television stories for CineLens.',
    'For every item return exactly one best-fit mood from: happy, sad, romantic, funny, scary, tense, calm, dark.',
    'If several moods seem possible, choose the single mood that best represents the overall viewing experience. Never return mixed.',
    'Choose the emotional experience supported by the plot, not genre alone.',
    'Every mood needs a short quotation or close paraphrase from the plot.',
    'Use lowercase mood values and omit unsupported moods. Return only JSON.',
    JSON.stringify(cleanItems)
  ].join('\n');
  const result = callGemini(prompt, schema, 12000);
  return {ok:true, model:MODEL, results:sanitizeMoodResults(result.results, cleanItems)};
}

function sanitizeMoodResults(results, cleanItems) {
  const allowed = {};
  cleanItems.forEach(function (item) { allowed[item.id] = true; });
  return (Array.isArray(results) ? results : []).filter(function (result) {
    return allowed[String(result && result.id)];
  }).map(function (result) {
    const seen = {};
    const moods = (Array.isArray(result.moods) ? result.moods : []).map(function (item) {
      return {mood:String(item && item.mood || '').toLowerCase(), confidence:Number(item && item.confidence), evidence:String(item && item.evidence || '').slice(0, 500)};
    }).filter(function (item) {
      if (!['happy','sad','romantic','funny','scary','tense','calm','dark'].includes(item.mood) || !item.evidence || !isFinite(item.confidence) || item.confidence < 0.55 || item.confidence > 1 || seen[item.mood]) return false;
      seen[item.mood] = true;
      return true;
    }).sort(function (left, right) { return right.confidence - left.confidence; }).slice(0, 1);
    return {id:String(result.id), moods:moods.length ? moods : [{mood:'calm', confidence:0.55, evidence:'The supplied plot presents a restrained emotional experience.'}]};
  });
}

function handleTitleTagging(request) {
  const items = Array.isArray(request.items)
    ? request.items.slice(0, MAX_ITEMS)
    : [];

  if (!items.length) {
    throw new Error('No titles supplied');
  }

  const minimumTags = clampNumber(
    request.minimumTags,
    1,
    DEFAULT_MAX_TAGS,
    DEFAULT_MIN_TAGS
  );

  const maximumTags = clampNumber(
    request.maximumTags,
    minimumTags,
    DEFAULT_MAX_TAGS,
    DEFAULT_MAX_TAGS
  );

  const cleanItems = items.map(function (item, index) {
    const existingTags = cleanTagList(item.existingTags, maximumTags);
    const excludedTags = cleanTagList(item.excludedTags, 200);

    const requestedAdditional = clampNumber(
      item.minimumAdditionalTags,
      0,
      maximumTags,
      Math.max(0, minimumTags - existingTags.length)
    );

    return {
      id: String(item.id || index),
      title: String(item.title || '').slice(0, 200),
      year: Number(item.year) || null,
      format: item.format === 'show' ? 'show' : 'movie',
      language: String(item.language || '').slice(0, 30),
      genres: Array.isArray(item.genres)
        ? item.genres.slice(0, 12).map(String)
        : [],
      plot: String(item.storyText || '').slice(0, MAX_STORY_CHARS),
      existingTags: existingTags,
      excludedTags: excludedTags,
      minimumAdditionalTags: requestedAdditional,
      preferredTagVocabulary: cleanTagList(
        item.preferredTagVocabulary,
        500
      )
    };
  });

  const prompt = buildTitleTaggingPrompt(
    cleanItems,
    minimumTags,
    maximumTags,
    request.continueTagging === true
  );

  const schema = {
    type: 'object',
    properties: {
      results: {
        type: 'array',
        items: {
          type: 'object',
          properties: {
            id: {
              type: 'string'
            },
            tags: {
              type: 'array',
              minItems: 1,
              maxItems: maximumTags,
              items: {
                type: 'object',
                properties: {
                  tag: {
                    type: 'string',
                    pattern: '^[a-z0-9]+(?:-[a-z0-9]+)*$'
                  },
                  confidence: {
                    type: 'number',
                    minimum: 0,
                    maximum: 1
                  },
                  evidence: {
                    type: 'string'
                  }
                },
                required: ['tag', 'confidence', 'evidence'],
                additionalProperties: false
              }
            },
            moods: {
              type: 'array',
              minItems: 1,
              maxItems: 1,
              items: {
                type: 'object',
                properties: {
                  mood: {
                    type: 'string',
                    enum: ['happy', 'sad', 'romantic', 'funny', 'scary', 'tense', 'calm', 'dark']
                  },
                  confidence: {
                    type: 'number',
                    minimum: 0,
                    maximum: 1
                  },
                  evidence: {
                    type: 'string'
                  }
                },
                required: ['mood', 'confidence', 'evidence'],
                additionalProperties: false
              }
            }
          },
          required: ['id', 'tags'],
          additionalProperties: false
        }
      }
    },
    required: ['results'],
    additionalProperties: false
  };

  // v91: the client now sends full 20-item batches (MAX_ITEMS) instead of 3-8.
  // 20 titles x up to 20 tags, each with an evidence quotation, does not fit in
  // the old 24k budget — the response would be truncated mid-JSON and surface
  // as "Gemini returned no result", which the client would then retry, turning
  // the throughput win into wasted quota.
  const result = callGemini(prompt, schema, 60000);

  return {
    ok: true,
    model: MODEL,
    results: sanitizeTaggingResults(
      result.results,
      cleanItems,
      maximumTags
    )
  };
}

function handleTagCloudNormalization(request) {
  const vocabulary = sanitizeVocabulary(request.tagVocabulary);

  if (vocabulary.length < 2) {
    throw new Error('At least two vocabulary tags are required');
  }

  const vocabularySet = {};

  vocabulary.forEach(function (item) {
    vocabularySet[item.tag] = true;
  });

  const prompt = [
    'You are consolidating the shared tag vocabulary for CineLens recommendations.',
    '',
    'Your task is to create an in-place rewrite plan, not an alias layer and not title tagging.',
    'Review the complete submitted vocabulary and frequencies.',
    'Return rewrite groups only when phrases are genuinely interchangeable recommendation signals.',
    '',
    'Rules:',
    '- The canonical value must be one existing tag from tagVocabulary.',
    '- Every tag in replace must also be an existing tag from tagVocabulary.',
    '- Do not invent tags.',
    '- Do not merge phrases merely because they share a word, prefix, modifier, setting, relationship, action, cause or outcome.',
    '- Keep materially different events separate. For example, accidental-poison and accidental-shooting are not interchangeable.',
    '- Merge clear synonyms, equivalent grammatical forms and redundant qualified variants only when they express the same recommendation signal.',
    '- Prefer the clearest and most reusable existing phrase as canonical.',
    '- Prioritize consolidating rare tags first: tags with lower count are more likely to be overly specific wording that should map into an existing clearer tag.',
    '- Prefer replacing a lower-count tag with a genuinely equivalent higher-count canonical tag when both are present.',
    '- Do not merge a common tag merely because a rare tag shares some words with it.',
    '- Omit uncertain mappings. Precision is more important than reducing the count.',
    '- Give each rewrite group a confidence from 0 to 1. Use 0.90 or higher only when every replaced tag is genuinely interchangeable with the canonical tag.',
    '',
    'Return rewriteGroups. Do not return identity mappings.',
    'Return only JSON matching the schema.',
    '',
    JSON.stringify({
      normalizationVersion: String(
        request.normalizationVersion || ''
      ),
      tagVocabulary: vocabulary,
      clientInstructions: String(request.instructions || '').slice(0, 4000)
    })
  ].join('\n');

  const schema = {
    type: 'object',
    properties: {
      rewriteGroups: {
        type: 'array',
        items: {
          type: 'object',
          properties: {
            canonical: {
              type: 'string',
              pattern: '^[a-z0-9]+(?:-[a-z0-9]+)*$'
            },
            replace: {
              type: 'array',
              minItems: 1,
              maxItems: 50,
              items: {
                type: 'string',
                pattern: '^[a-z0-9]+(?:-[a-z0-9]+)*$'
              }
            },
            confidence: {
              type: 'number',
              minimum: 0,
              maximum: 1
            }
          },
          required: ['canonical', 'replace', 'confidence'],
          additionalProperties: false
        }
      }
    },
    required: ['rewriteGroups'],
    additionalProperties: false
  };

  const result = callGemini(prompt, schema, 24000);

  return {
    ok: true,
    model: MODEL,
    rewriteGroups: validateRewriteGroups(
      result.rewriteGroups,
      vocabularySet
    )
  };
}

function callGemini(prompt, schema, maxOutputTokens) {
  const apiKey = PropertiesService
    .getScriptProperties()
    .getProperty('GEMINI_API_KEY');

  if (!apiKey) {
    throw new Error('GEMINI_API_KEY is missing');
  }

  const url =
    'https://generativelanguage.googleapis.com/v1beta/models/' +
    MODEL +
    ':generateContent';

  const response = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    headers: {
      'x-goog-api-key': apiKey
    },
    muteHttpExceptions: true,
    payload: JSON.stringify({
      contents: [{
        role: 'user',
        parts: [{
          text: prompt
        }]
      }],
      generationConfig: {
        temperature: 0.1,
        maxOutputTokens: maxOutputTokens || 24000,
        responseMimeType: 'application/json',
        responseJsonSchema: schema
      }
    })
  });

  const status = response.getResponseCode();
  const body = JSON.parse(response.getContentText() || '{}');

  if (status < 200 || status >= 300) {
    throw new Error(
      body.error && body.error.message
        ? body.error.message
        : 'Gemini request failed: HTTP ' + status
    );
  }

  const text =
    body.candidates &&
    body.candidates[0] &&
    body.candidates[0].content &&
    body.candidates[0].content.parts &&
    body.candidates[0].content.parts[0] &&
    body.candidates[0].content.parts[0].text;

  const finishReason =
    body.candidates &&
    body.candidates[0] &&
    body.candidates[0].finishReason;

  if (!text) {
    throw new Error(
      'Gemini returned no result' +
      (finishReason ? ' (finishReason: ' + finishReason + ')' : '')
    );
  }

  // A truncated response is not a malformed one — say which it was, so a batch
  // that outgrows the output budget is obvious instead of looking like a
  // random model failure.
  if (finishReason === 'MAX_TOKENS') {
    throw new Error(
      'Gemini response hit the output token limit; reduce the batch size'
    );
  }

  return JSON.parse(text);
}

function sanitizeTaggingResults(results, cleanItems, maximumTags) {
  const allowedIds = {};

  cleanItems.forEach(function (item) {
    allowedIds[item.id] = {
      excluded: toLookup(item.excludedTags),
      existing: toLookup(item.existingTags)
    };
  });

  return (Array.isArray(results) ? results : [])
    .filter(function (result) {
      return allowedIds[String(result && result.id)];
    })
    .map(function (result) {
      const id = String(result.id);
      const rules = allowedIds[id];
      const seen = {};

      const tags = (Array.isArray(result.tags) ? result.tags : [])
        .map(function (item) {
          return {
            tag: normalizeTag(item && item.tag),
            confidence: Number(item && item.confidence),
            evidence: String(item && item.evidence || '').slice(0, 500)
          };
        })
        .filter(function (item) {
          if (
            !item.tag ||
            !item.evidence ||
            !isFinite(item.confidence) ||
            item.confidence < 0 ||
            item.confidence > 1 ||
            seen[item.tag] ||
            rules.excluded[item.tag] ||
            rules.existing[item.tag]
          ) {
            return false;
          }

          seen[item.tag] = true;
          return true;
        })
        .slice(0, maximumTags);

      const moods = (Array.isArray(result.moods) ? result.moods : [])
        .map(function (item) {
          return {
            mood: String(item && item.mood || '').toLowerCase(),
            confidence: Number(item && item.confidence),
            evidence: String(item && item.evidence || '').slice(0, 500)
          };
        })
        .filter(function (item, index, values) {
          return ['happy', 'sad', 'romantic', 'funny', 'scary', 'tense', 'calm', 'dark'].indexOf(item.mood) >= 0 &&
            item.evidence && isFinite(item.confidence) && item.confidence >= 0.55 && item.confidence <= 1 &&
            values.findIndex(function (value) { return value.mood === item.mood; }) === index;
        })
        .sort(function (left, right) { return right.confidence - left.confidence; })
        .slice(0, 1);

      return {
        id: id,
        tags: tags,
        moods: moods.length ? moods : [{
          mood: 'calm',
          confidence: 0.55,
          evidence: 'The supplied plot presents a restrained emotional experience.'
        }]
      };
    });
}

function sanitizeVocabulary(value) {
  const counts = {};

  (Array.isArray(value) ? value : [])
    .slice(0, MAX_VOCABULARY_TAGS)
    .forEach(function (item) {
      const tag = normalizeTag(item && item.tag);

      if (!tag) {
        return;
      }

      counts[tag] = Math.max(
        counts[tag] || 0,
        Math.max(1, Number(item.count) || 1)
      );
    });

  return Object.keys(counts)
    .map(function (tag) {
      return {
        tag: tag,
        count: counts[tag]
      };
    })
    .sort(function (a, b) {
      return a.count - b.count || a.tag.localeCompare(b.tag);
    });
}

function validateRewriteGroups(groups, vocabularySet) {
  const output = [];
  const assigned = {};

  (Array.isArray(groups) ? groups : []).forEach(function (group) {
    const canonical = normalizeTag(group && group.canonical);
    const confidence = Number(group && group.confidence);

    if (
      !canonical ||
      !vocabularySet[canonical] ||
      !isFinite(confidence) ||
      confidence < 0.9
    ) {
      return;
    }

    const replacements = cleanTagList(
      group && group.replace,
      50
    ).filter(function (sourceTag) {
      if (
        sourceTag === canonical ||
        !vocabularySet[sourceTag] ||
        assigned[sourceTag]
      ) {
        return false;
      }

      assigned[sourceTag] = canonical;
      return true;
    });

    if (replacements.length) {
      output.push({
        canonical: canonical,
        replace: replacements,
        confidence: confidence
      });
    }
  });

  return output;
}

function cleanTagList(value, limit) {
  const seen = {};

  return (Array.isArray(value) ? value : [])
    .map(normalizeTag)
    .filter(function (tag) {
      if (!tag || seen[tag]) {
        return false;
      }

      seen[tag] = true;
      return true;
    })
    .slice(0, limit || 200);
}

function normalizeTag(value) {
  const tag = String(value || '')
    .toLowerCase()
    .trim()
    .replace(/\s+/g, '-');

  return /^[a-z0-9]+(?:-[a-z0-9]+)*$/.test(tag)
    ? tag
    : '';
}

function toLookup(values) {
  const output = {};

  (values || []).forEach(function (value) {
    output[value] = true;
  });

  return output;
}

function clampNumber(value, min, max, fallback) {
  const number = Number(value);

  if (!isFinite(number)) {
    return fallback;
  }

  return Math.max(min, Math.min(max, Math.round(number)));
}

function jsonOutput(value) {
  return ContentService
    .createTextOutput(JSON.stringify(value))
    .setMimeType(ContentService.MimeType.JSON);
}
